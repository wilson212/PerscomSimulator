	/// Gets a selection factor rating which is used to determine who is more/less
	/// likely to get selected for a lateral position based off of many individual factors.
	public SoldierWrapper.GetLateralSelectionFactor(Position) : int (0-100)
	{
		> Lateral Promotion (Higher Ambition soldier's more likely as Next.Position.Stature is higher):
			-> Factor in Soldier.Ambition vs (Current.Position.Desirability and Next.Position.Desirability) (Prestiege) [30 points]
				"A high ambition will want lower stature's less, and higher stature's more"
				[This is a primary factor... does the soldier even want to move?]
			-> Factor in weighted skills (will they do well?) [30 points]
			---> Get "Form" rating (0.0 -> 9.9) for Current position vs Other position
					"A rating indicating how well they will do in this position"
			-> Factor in time until retirement [10 points]
				"If close to retirement, they may not want another position change"
			-> Factor in time in current position [20 points]
				"Score caps at 10 points if locked in position"
			-> Factor in time in grade (seniority, uses ambition) [10 points]
				"A higher time in grade will warrant a promotion over a lateral position change"
	}

	public SoldierWrapper.WillAttemptSpecialAssignment(Position) : bool
	{
		> Special Assignment / Cross-corps selection (Higher Ambition soldier's more likely, regardless of stature):
			-> Factor in Soldier.Ambition [60 points]
				"A higher ambition will want the promotion more, regardless of the Next.Position.Desirability"
				[This is a primary factor... does the soldier even want to move?]
			-> Factor in weighted skills (will they do well?)
			-> Factor in if School is required, and how much time they need to school compared to retirement date
			-> Factor in Unit tier level
			---> Get "Form" rating (0.0 -> 9.9)
					"A rating indicating how well they will do in this position. Less important with higher ambition"
			-> Factor in time until retirement [20 points]
				"If close to retirement, they may not want another position change"
	}

public Soldier SelectCandidate(this Position targetPosition, IEnumerable<Soldier> activeRoster, IterationDate currentDate)
{
    var targetRank = targetPosition.Billet.Rank;

    // ---------------------------------------------------------
    // STEP 1: LATERAL MOVE EVALUATION
    // ---------------------------------------------------------
    var lateralPool = activeRoster
        .Where(s => s.Rank.Id == targetRank.Id) // Must be same rank
        .Where(s => s.GetTimeInBillet(currentDate) > 6) // Minimum time before allowed to transfer
        .ToList();

    if (lateralPool.Any())
    {
        // Score every eligible soldier based on your 0-100 factor
        var bestLateralCandidate = lateralPool
            .Select(s => new 
            { 
                Soldier = s, 
                Score = s.GetLateralSelectionFactor(targetPosition) 
            })
            .Where(x => x.Score > 50) // Minimum threshold to even consider moving
            .OrderByDescending(x => x.Score)
            .FirstOrDefault();

        if (bestLateralCandidate != null)
            return bestLateralCandidate.Soldier;
    }

    // ---------------------------------------------------------
    // STEP 2: PROMOTION BOARD EVALUATION
    // ---------------------------------------------------------
    var eligibleForPromotion = activeRoster
        .Where(s => s.Rank.PayGrade == targetRank.PayGrade - 1)
        .Where(s => s.GetTimeInGrade(currentDate) >= targetRank.MinimumTIG)
        .ToList();

    // The gatekeeper: Do they even want the job?
    var volunteers = eligibleForPromotion
        .Where(s => s.WillAttemptGradeIncrease())
        .ToList();

    if (volunteers.Any())
    {
        // Run your stat weights and Form Rating to pick the winner
        return RunEvaluationBoard(volunteers, targetPosition);
    }

    // ---------------------------------------------------------
    // STEP 3: NO CANDIDATES
    // ---------------------------------------------------------
    return null; // Position remains empty
}

// 1. THE ATTENDANCE ROLL
// E.g., 20 Conscientiousness = 95% chance to show up. 1 Conscientiousness = 30%.
bool didAttend = _random.Next(1, 101) <= (25 + (soldier.Conscientiousness * 3.5));

if (didAttend)
{
    // 2. CALCULATE RESISTANCE (Max: 10)
    // Composure (handles immediate stress) + Adaptability (handles changing game scenarios)
    double resistance = (soldier.Composure + soldier.Adaptability) / 4.0;

    // 3. CALCULATE UNIT DRAMA (Max: +5 Stress)
    // We blend the Leader's personality with their tactical coordination.
    // 40 is max possible (20 Agreeableness + 20 Teamwork). 
    double leaderFriction = (40.0 - (squadLeader.Agreeableness + squadLeader.Teamwork)) / 8.0;
    
    // A soldier with high Teamwork is a "team player" and absorbs some of that friction.
    // A soldier with low Teamwork gets annoyed easily by bad leadership.
    double personalDramaBuffer = soldier.Teamwork / 10.0; // Grants up to a 2.0 reduction
    
    double finalDramaPenalty = Math.Max(0, leaderFriction - personalDramaBuffer);

    // 4. THE BURNOUT DELTA
    // A stressful job plus a bad boss, minus their personal mental fortitude.
    double burnoutGain = (currentPosition.StressLevel + finalDramaPenalty) - resistance;

    // Minimum gain of 1.0 just so they don't play literally forever without *some* fatigue.
    soldier.Burnout += Math.Max(1.0, burnoutGain); 
}
else
{
    // 5. THE RECOVERY
    // They skipped the weekend op to hang out with their family. 
    // They drop a flat chunk of Burnout.
    soldier.Burnout -= 15.0; 
}

// 6. CLAMP IT
soldier.Burnout = Math.Clamp(soldier.Burnout, 0.0, 100.0);
